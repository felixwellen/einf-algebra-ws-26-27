
\subsection{Grundlegende Definitionen}

\begin{definition}
  Sei $M$ eine Menge. Eine \begriff{Verknüpfung} (auf \(M\)) ist eine Abbildung $\_\cdot\_:M\times M \to M$.\footnote{Die Unterstriche ``\(\_\)'' im Namen ``\(\_\cdot\_\)'' bedeuten, dass wir Anwendungen der Abbildung ``\(\_\cdot\_:M\times M\to M\)'' auf Elemente \((x,y)\in M\) schreiben als ``\(x\cdot y\)''. Im Gegensatz dazu würden wir ja für \(f:M\times M \to M\) die Anwendung auf \((x,y)\) schreiben als \(f((x,y))\).}
\end{definition}

\begin{definition}
  Sei \(M\) eine Menge und \(\_\cdot\_\) eine Verknüpfung auf \(M\).
  \begin{enumerate}
  \item $\_\cdot\_$ heißt \begriff{assoziativ}, wenn gilt:
    \begin{align*}
      \forall x,y,z\in M. (x\cdot y)\cdot z = x\cdot (y\cdot z)
    \end{align*}
    Das Paar \((M,\cdot)\) heißt in diesem Fall \begriff{Halbgruppe}.
  \item Ein Element $e\in M$ heißt \begriff{neutrales Element}, wenn für alle \(x\in M\) gilt: \(x=e\cdot x=x\cdot e\).
    Eine Halbgruppe $(M,\cdot)$ zusammen mit einem neutralen Element $e\in M$ heißt \begriff{Monoid}.
  \item Sei $(M,\cdot,e)$ ein Monoid und $x,y\in M$. $y$ heißt \begriff{invers} zu $x$, wenn $y\cdot x=x\cdot y=e$ gilt.
    Ein Monoid $(M,\cdot,e)$ heißt \begriff{Gruppe}, wenn es zu jedem Element von \(M\) ein Inverses gibt.
  \item Eine Verknüpfung \(\_\cdot\_\) auf einer Menge \(M\) heißt \begriff{kommutativ}, wenn für alle \(x,y\in M\) gilt: \(x\cdot y=y\cdot x\).
    Eine Gruppe \(G\) mit Verknüpfung \(\_\cdot\_\) heißt \begriff{abelsche Gruppe}, wenn \(\_\cdot\_\) kommutativ ist.
  \end{enumerate}
\end{definition}

\begin{beispiel}
  \begin{enumerate}[(a)]
  \item \(\N\), was bei uns stets die 0 einschließt, ist ein Monoid mit Neutralelement 0 und Verknüpfung \(+\). \(N\) ist aber auch ein Monoid mit Neutralelement 1 und Verknüpfung \(\cdot\).
  \item \(\Z,\Q,\R,\C\) sind abelsche Gruppen mit der bekannten Verknüpfung \(+\).
  \item \(\{1,-1\},\Q\setminus\{0\},\R\setminus\{0\},\C\setminus\{0\}\) sind Gruppen mit der Verknüpfung \(\cdot\).
  \item Sei \(M\) eine Menge. Die Bijektionen \(M\to M\) bilden eine Gruppe, die wir \begriff{Permutationsgruppe} oder \begriff{symmetrische Gruppe} nennen und mit \(S(M)\)\index{\(S(M)\)} bezeichnen.
  \item Sei \(V\) ein \(k\)-Vektorraum, dann bilden die invertierbaren linearen Abbildungen \(V\) mit der Komposition (bzw.\ Matrizenmultiplikation) die Gruppe \(\GL(n,k)\). Weiter gibt es die aus der linearen Algebra bekannten Matrizengruppen \(O_n\) und \(SO_n\).
  \end{enumerate}
\end{beispiel}

Es gibt die zusätzlichen Unterscheidungen zwischen links- und rechtsneutral bzw.\ links- und rechtsinvers. Aufgrund des Sachverhalts in (a) und (b) wird es sich nicht für uns lohnen, diese Begriffe zu definieren:

\begin{bemerkung}
  Sei \(H\) eine Menge mit Verknüpfung \(\_\cdot\_\).
  \begin{enumerate}[(a)]
  \item Sei \(e_l\in H\) sodass \(e\cdot x=x\) und \(e_r\in H\) sodass \(x\cdot e_r=x\) für alle \(x\) gelten. Dann gilt \(e_l=e_r\). Damit gibt es insbesondere in einer Halbgruppe höchstens ein Neutralelement.
  \item Sei \(H\) nun Monoid mit neutralem Element \(e\)
    und \(x,y,y^\prime\in H\) sodass \(xy=e\) und \(y^\prime x=e\) dann gilt \(y=y^\prime\). Damit hat in einem Monoid jedes Element höchstens ein Inverses.
  \item Sei \(H\) Gruppe, dann gilt für alle \(g\in H\): \((g^{-1})^{-1}=g\).
    Weiter gilt für \(g,h\in H\): \((gh)^{-1}=h^{-1}g^{-1}\).
  \item
   Sei nun \(H\) Halbgruppe und existiere \(e\in H\) sodass gilt:
   \begin{enumerate}[(i)]
   \item \(ex=x\) für alle \(x\in H\) 
   \item und für jedes \(x\in H\) ein \(y\in H\) existiert mit \(yx=e\).
   \end{enumerate}
   Dann ist \(H\) bereits eine Gruppe mit Neutralement \(e\).
  \end{enumerate}
\end{bemerkung}

\begin{beweis}
  \begin{enumerate}[(a)]
  \item \(e_l=e_l\cdot e_r=e_r\).
  \item \(y=e\cdot y=(y^\prime\cdot x) \cdot y=y^\prime\cdot (x \cdot y)=y^\prime\cdot e=y^\prime\).
  \item
    Die beiden Bedingungen implizieren, dass wir stets wie folgt kürzen können:
    \[
      ax=ay \Rightarrow x=y
    \]
    Sei \(x\) beliebig und \(y\) sodass \(yx=e_l\).
    Dann ist \(y\) invers zu \(x\): \(xyxy=x(yx)y=xy \Rightarrow xy=e_l\) .
  \item Übungsblatt 1.
  \end{enumerate}
\end{beweis}

\begin{konvention}
  \begin{enumerate}[(i)]
  \item Bei multiplikativen Gruppen \(G\), also Gruppen deren Verknüpfung mit \(\_\cdot\_\) bezeichnet wird, schreiben wir \(x^{-1}\) für das Inverse von \(x\in G\) und 1 für das Neutralelement. ``\(\cdot\)'' kann auch weggelassen werden.
  \item Für die Verknüpfung in abelschen Gruppen \(A\) verwenden wir oft \(+\).
    In diesem Fall schreiben wir \(-x\) für das Inverse von \(x\in A\) und \(0\) für das Neutralelement.
  \item In Halbgruppen kann die Klammerung von Verknüpfungsausdrücken weggelassen werden.
  \item Sei \((G,\cdot,e)\) Gruppe und \(g_0,\dots,g_n\in G\). Wir schreiben\index{\(\prod_{i=0}^na_i\)}
    \[
    \prod_{i=0}^na_i:= a_0\cdot a_1\cdot \cdots \cdot a_n
    \]
    wobei im Fall \(n<0\) gilt: \(\prod_{i=0}^na_i=e\).
  \item Analog verwenden wir in abelschen Gruppen die bekannte Schreibweise\index{\(\sum_{i=0}^na_i\)}
    \[
      \sum_{i=0}^na_i=a_0+\dots+a_n
    \]
  \item Sei \(H\) Halbgruppe und \(U,V\subseteq H\) und \(a\in H\). Wir schreiben:
    \begin{align*}
      a\cdot U&:=\{a\cdot x\mid x\in U\}\subseteq H \\
      U\cdot V&:=\{x\cdot y\mid x\in U, y\in V\}\subseteq H\text{ und }\\
      U^{-1}&:=\{x^{-1}\mid x\in U\}\text{ falls alle \(x\in U\) invertierbar sind.}
    \end{align*}
  \end{enumerate}
\end{konvention}

\subsection{Untergruppen}

\begin{definition}
  Sei \(M\) Monoid und \(U\subseteq M\) eine Teilmenge.
  \begin{enumerate}[(a)]
  \item \(U\) heißt \begriff{Untermonoid}, wenn \(e\in U\) gilt und für alle \(x,y\in U\) stets \(xy\in U\) gilt.
  \item Ist \(M\) zusätzlich eine Gruppe, \(U\) ein Untermonoid und gilt zusätzliche \(x^{-1}\in U\) für alle \(x\in U\), so heißt \(U\) \begriff{Untergruppe}.
  \end{enumerate}
\end{definition}

\begin{lemma}
  \begin{enumerate}
  \item Untergruppen sind Gruppen.
  \item Seien \(G\) Gruppe und \(U\subseteq G\) eine nichtleere Teilmenge.
  Weiter sei für alle \(g,h\in U\) auch \(gh^{-1}\in U\).
  Dann ist \(U\) eine Untergruppe.
  \end{enumerate}
\end{lemma}

\begin{beweis}
  Übung?
\end{beweis}

\begin{beispiel}
  \label{untergruppen-von-ZZ-und matrizengruppen}
  \begin{enumerate}[(a)]
  \item Die Teilmengen der Form \(n\cdot \Z\subseteq \Z\) für \(n\in\Z\) sind genau die Untergruppen der Gruppe \((\Z,+)\).
  \item \(\O_n\subseteq \GL_n(\R)\), \(\SO_n\subseteq \GL_n(\R)\) und \(\SL_n\subseteq \GL_n(k)\), wobei \(\SL_n:=\{M\in k^{n\times n}\mid \det(M)=1\}\).
  \end{enumerate}
\end{beispiel}

\subsection{Homomorphismen}

\begin{definition}
  Seinen \((M,\cdot_1,e_1)\) und \((N,\cdot_2,e_2)\) Monoide und \(f:M\to N\) eine Abbildung.
  \begin{enumerate}[(a)]
  \item \(f\) heißt \begriff{Monoidhomomorphismus}, wenn gilt:
    \begin{align*}
      f(e_1)=e_2\text{ und für alle \(x,y\in M\) gilt } f(x\cdot_1 y)=f(x)\cdot_2 f(y).
    \end{align*}
  \item Seien \(M,N\) Gruppen und \(f\) Monoidhomomorphismus. Dann heißt \(f\) \begriff{Gruppenhomomorphismus}. Wir müssen nicht fordern, dass für alle \(x\in G\) gilt: \(f(x^{-1})=f(x)^{-1}\), weil sich das zeigen lässt. Siehe dazu die nächste Bemerkung.
  \item Ein Gruppenhomomorphismus \(f:G\to H\) heißt \begriff{Gruppenisomorphismus},
    wenn es \(g:H\to G\) gibt, sodass \(g\circ f=\id_G\) und \(f\circ g=\id_H\) gelten. Das ist genau dann der Fall, wenn \(f\) bijektiv ist.
  \end{enumerate}
\end{definition}

\begin{bemerkung}
  Sei \(f:G\to H\) eine Abbildung zwischen Gruppen \((G,\cdot_1,e_1),(H,\cdot_2,e_2)\) und gelte \(\forall x,y\in G. f(x\cdot_1y)=f(x)\cdot_2 f(y)\), dann gilt:
  \begin{align*}
    f(e_1)=e_2\text{ und für alle \(x\in G\) gilt }f(x^{-1})=f(x)^{-1}.
  \end{align*}
\end{bemerkung}

\begin{beweis}
  \(f(e_1)=f(e_1e_1)=f(e_1)f(e_1)\) also \(e_2=f(e_1)\). \(e_2=f(xx^{-1})=f(x)f(x^{-1})\), also \(f(x^{-1})=f(x)^{-1}\) wegen Eindeutigkeit von Inversen.
\end{beweis}

\begin{beispiel}
  \label{bsp-homs}
  \begin{enumerate}[(a)]
  \item Sei \(M\) Monoid und \(x\in M\), dann ist \(n\mapsto x^n:\N\to M\) ein Monoidhomomorphismus.
  \item Sei \(G\) Gruppe und \(x\in G\), dann ist \(k\mapsto x^k:\Z\to G\) ein Gruppenhomomorphismus, wobei wir für \(k<0\) festlegen: \(x^k:=(x^{|k|})^{-1}\).
  \item Aus der Analysis ist der Gruppenisomorphismus \(e:(\R,+)\to (\R_{>0},\cdot)\) bekannt.
  \item\label{bsp-translation} Sei \((G,\cdot)\) eine Gruppe und \(g\in G\).
    Die Abbildung \(\tau_g:= h\mapsto gh :G\to G\) heißt \begriff{Linkstranslation} (mit \(g\)). Das liefert den Gruppenhomomorphismus \(g\mapsto \tau_g: (G,\cdot)\to (S(G),\circ)\).
  \end{enumerate}
\end{beispiel}

\begin{konvention}
  Sei \(G\) eine Gruppe und \(H\subseteq G\)
  Wir schreiben \(H<G\), wenn \(H\) eine Untergruppe von \(G\) ist.
\end{konvention}

\begin{lemma}
  Seien \((G,\cdot,e_G)\) und \((H,\cdot,e_H\) Gruppen.
  Sei \(f:G\to H\) Gruppenhomomorphismus.
  \begin{enumerate}[(a)]
  \item Es sind
  \begin{align*}
    &\Bild(f):=f(G)< H \text{ und }\\
    &\Kern(f):=\{x\in G\mid f(x)=e_H\}< G
  \end{align*}
  jeweils Untergruppen.
  \item \(f\) ist genau dann injektiv, wenn \(\Kern(f)=\{e_G\}\) gilt.
  \item Jede Gruppe ist (bis auf Isomorphismus) Untergruppe einer Permutationsgruppe.
  \end{enumerate}
\end{lemma}

\begin{beweis}
  \begin{enumerate}[(a)]
  \item \(e_2\in\Bild(f)\) und für \(x,y\in\Bild(f)\) gib es Urbilder \(u,v\in G\)
    mit \(f(u)=x, f(v)=y\) also gilt: \(f(uv^{-1})=f(u)f(v)^{-1}=xy^{-1}\in\Bild(f)\).
    Rest: Übung.
  \item Sei \(\Kern(f)=\{0\}\). \\
    Dann gilt: \(f(x)=f(y) \Rightarrow f(xy^{-1})=e_2 \Rightarrow xy^{-1}\in\Kern(f)=\{0\} \Rightarrow xy^{-1}=e_1 \Rightarrow x=y\). 
  \item Sei \((G,\cdot,e)\) eine Gruppe und \(f:=g\mapsto \tau_g:(G,\cdot)\to (S(G),\circ)\) wie in \ref{bsp-homs} \ref{bsp-translation}. Dann gilt \(\Kern(f)=\{e\}\), also ist \(f\) injektiv.
  \end{enumerate}
\end{beweis}

\subsection{Faktorgruppen}

\begin{konvention}
  \(|M|\) bezeichnet die Anzahl der Elemente einer Menge \(M\).
  Für unendliche Mengen setzen wir \(|M|=\infty\) und verwenden die Rechenregeln \(x\cdot \infty=\infty=\infty\cdot x\) für \(x\in\N\cup\{\infty\}\).
\end{konvention}

\begin{lemma}
  \label{lemma-zu-nebenklassen}
  Sei \(G\) eine Gruppe, \(H\subseteq G\) Untergruppe und \(g\in G\).
  Eine Teilmenge der Form
  \[gH\coloneq \{gh\in G\mid h\in H\}\]
  nennen wir \begriff{Linksnebenklasse} und entsprechend \(Hg\coloneq\{hg\mid h\in H\}\) \begriff{Rechtsnebenklasse}. Es gilt:
  \begin{enumerate}[(a)]
  \item\label{nbkl-aequivrel} \(xH=yH\) genau dann wenn \(xH\cap yH\neq \emptyset\) genau dann wenn  \(y^{-1}x\in H\). 
  \item \(G\) ist die disjunkte Vereinigung der Linksnebenklassen von \(H\).
  \item Zwischen je zwei Linksnebenklassen gibt es eine Bijektion.
  \end{enumerate}
\end{lemma}
\rednote{Stand nach VL vom 20.10.2026: b bewiesen, aber a noch nicht.}
\begin{beweis}
  \begin{enumerate}[(a)]
  \item Zuerst die zweite Äquivalenz: \(xH\cap yH\neq\emptyset \Leftrightarrow
    \exists u,v\in H. xu=yv
    \Leftrightarrow \exists u,v\in H. y^{-1}x=vu^{-1}
    \Leftrightarrow y^{-1}x\in H\)
  \item Folgt aus \cref{nbkl-aequivrel}, mit der Festellung, dass jedes \(g\in G\) in einer Linksnebenklasse enthalten ist (nämlich in \(gH\)).
  \item \ref{bsp-homs} \ref{bsp-translation} liefert Bijektion \(H\to gH\) für jedes \(g\in G\).
  \end{enumerate}
\end{beweis}

\begin{definition}
  Sei \(G\) eine Gruppe und \(H<G\).
  \begin{enumerate}[(a)]
  \item Wir bezeichnen mit \(G/H\)\index{\(G/H\)} die Menge der Linksnebenklassen von \(H\).
  \item Die Zahl \((G : H)\coloneq |G/H|\)\index{\((G : H)\)} heißt \begriff{Index} von \(H\) in \(G\).
  \item Wir nennen \(\ord(G)\coloneq |G|\)\index{\(\ord\)} die \begriff{Ordnung} von \(G\).
  \end{enumerate}
\end{definition}

\begin{satz}[Satz von Lagrange]
  Sei \(G\) eine Gruppe und \(H\) eine Untergruppe von \(G\), dann gilt:
  \begin{align*}
    |G|&=|G/H |\cdot |H| \\
    \llap{\text{bzw.\quad }}\ord(G)&=(G : H)\cdot \ord(H)
  \end{align*}
\end{satz}

\begin{beweis}
  Folgt aus \ref{lemma-zu-nebenklassen}.
\end{beweis}

\begin{beispiel}
  Sei \(U\coloneq n\Z\subseteq \Z\) mit \(n\in\N, n>0\) eine der aus \Cref{untergruppen-von-ZZ-und matrizengruppen} bekannten Untergruppen.
  Dann ist jede Linksnebenklasse identisch zu genau einer der folgenden Linksnebenklassen:
  \[0+U, 1+U,\dots, (n-1)+U. \]
  \textbf{Beobachtung: } \(\Z/n\Z\) ist für \(n\in\N\) mit Addition von Linksnebenklassen also \((k+U)+^\prime (l+U)\coloneq (k+l)+U\) wieder eine Gruppe.
\end{beispiel}

Im Allgemeinen Fall muss eine Untergruppe \(H<G\) ein \emph{Normalteiler} sein, damit \(G/H\) auf eine natürliche Art wieder eine Gruppe ist:

\begin{definition}
  Sei \(G\) eine Gruppe und \(N\subseteq G\) Untergruppe.
  \(U\) heißt \begriff{Normalteiler}, wenn für alle \(g\in G\) gilt:
    \[gN=Ng.\]
    Äquivalent können wir fordern, dass für alle \(g\in G\) und \(n\in N\) gilt: \(gng^{-1}\in N\).
    Wir schreiben in diesem Fall \(N\triangleleft G\)\index{\(N\triangleleft G\)}.
\end{definition}

\begin{beispiel}
  \begin{enumerate}[(a)]
  \item Sei \(A\) abelsche Gruppe. Dann ist jede Untergruppe auch Normalteiler.
  \item Elemente \(\pi\in S_3=S(\{1,2,3\})\) notieren wir in der Form:
    \[\left(
    \begin{array}{ccc}
      1 & 2 & 3 \\
      \pi(1) & \pi(2) & \pi(3)
    \end{array}
    \right)\]
    Sei \(\tau_{12}=\left(
    \begin{array}{ccc}
      1 & 2 & 3 \\
      2 & 1 & 3
    \end{array}
    \right)\). Dann ist die Teilmenge \(\{\id,\tau_{12}\}\) eine Untergruppe, die kein Normalteiler ist.
  \item Sei \(f:G\to H\) eine Gruppenhomomorphismus. Dann ist \(\Kern(f)\) ein Normalteiler in \(G\). Um das zu sehen, halten wir zunächst fest, dass eine beliebige Untergruppe \(N<G\) genau dann ein Normalteiler ist, wenn \(N=gNg^{-1}\) für alle \(g\in G\) gilt. Es reicht also zu zeigen, dass für alle \(g\in G\) und \(x\in \Kern(f)\) gilt: \(gxg^{-1}\in \Kern(f)\). Das ist einfach so, weil: \(f(gxg^{-1})=f(g)f(x)f(g^{-1})=e\), also \(gxg^{-1}\in\Kern(f)\).
  \end{enumerate}
\end{beispiel}

\begin{definition}
  Sei \(G\) Gruppe und \(N\subseteq G\) Normalteiler.
  \begin{enumerate}[(i)]
  \item Durch
    \[
    gN\cdot hN\coloneq ghN
    \]
    ist eine Verknüpfung auf \(G/N\) gegeben,
    sodass \(G/N\) mit dieser Verknüpfung eine Gruppe ist.
    Wir nennen \(G/N\) mit dieser Verknüpfung \begriff{Faktorgruppe}.
  \item Die Abbildung \(\pi\coloneq g\mapsto gN:G\to G/N\) ist ein Gruppenhomomorphismus.
  \end{enumerate}
\end{definition}

\begin{beweis}
  \begin{enumerate}[(i)]
  \item \((aN) \cdot (bN)=\{a\}\cdot N\cdot bN=\{a\}\cdot Nb \cdot N=\{a\}\cdot bN \cdot N=\{ab\}\cdot N \cdot N=abN\)
  \item \(\pi(gh)=\pi(g)\pi(h)\).
  \end{enumerate}
\end{beweis}

  Im Fall \(N=\Kern(f)\) für Gruppenhomomorphismus \(f:G\to H\), ist \(G/\Kern(f)\) isomorph zu \(\Bild(f)\).
Das zeigen wir gleich allgemeiner:

\begin{satz}[Homomorphiesatz]
  Es seien \(f:G\to H\) ein Gruppenhomomorphismus und \(N\subseteq G\) ein Normalteiler mit \(N\subseteq \Kern(f)\).
  Dann existiert ein eindeutiger Gruppenhomomorphismus \(\bar{f}:G/N\to H\)
  mit \(\bar{f}\circ\pi=f\):
  \begin{center}
    \begin{tikzcd}
      G\ar[rr,"f"]\ar[rd,"\pi",swap] & & H \\
       & G/N\ar[ru,"\bar{f}",swap] &
    \end{tikzcd}
  \end{center}
  Es gilt: \(\Bild(f)=\Bild(\bar{f})\), \(\Kern(\bar{f})=\pi(\Kern(f))\)
  und \(\Kern(f)=\pi^{-1}(\Kern(\bar{f}))\).
\end{satz}
